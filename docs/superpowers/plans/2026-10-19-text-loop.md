# Text Loop (M1 + M3) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A typed utterance goes through understanding → conversation state → hard-filtered Postgres search → explainable ranking → validated top-3 response, including corrections, zero-result relaxation and follow-ups, measured by a live eval harness at ≥90%.

**Architecture:** Deterministic pure-Python components (number parser, filter builder, ranker, relaxation prober, state reducer, validator, templates), wired by a thin LangGraph graph. The LLM (Groq primary, Gemini fallback via LangChain wrappers in `providers/`) does only two jobs: fill a typed `TurnIntent` and phrase a structured response from a fact sheet. Postgres holds the catalog, aliases (pg_trgm normalisation) and LangGraph checkpoints. Voice, API and UI come in Plan 2; this plan exposes a `TurnRunner.run(session_id, text)` seam that they will call.

**Tech Stack:** Python 3.12, uv, Pydantic v2, pydantic-settings, psycopg 3 (async) + psycopg-pool, Postgres 16 + pg_trgm, LangGraph (+ langgraph-checkpoint-postgres), langchain-groq, langchain-google-genai, PyYAML, pytest + pytest-asyncio, ruff, mypy.

**Spec:** `docs/decisions.md` (why) and `docs/architecture.md` (how). Also follow `CLAUDE.md`, `docs/standards.md` and `docs/definition-of-done.md`.

## Global Constraints

- Python `>=3.12`, managed with `uv`. Dependencies are added with `uv add`, never by hand-editing `uv.lock`.
- Money is `int` INR; weights are `int` kg. No floats for money.
- The LLM never writes SQL. SQL is built only in `src/voice_agent/catalog/filters.py`, parameterized.
- Hard filters: `budget_min, budget_max, body_type, fuel, city, make, model, min_payload_kg, max_km, min_year`. `purpose` and `budget_target` are never filters.
- "under / ke andar / tak / max X" → `price <= X`; "X to Y" → between; "around X" → `budget_max = round(1.1·X)`, `budget_target = X`; a bare amount ("budget 5 lakh") → `budget_max`.
- A correction changes exactly one slot.
- Vendor SDKs (langchain-*, psycopg) are imported only in `src/voice_agent/providers/` and `src/voice_agent/db/`; LangGraph only in `src/voice_agent/graph/`.
- Every LLM-composed response passes `response/validator.py`, else it is regenerated once, else replaced by `response/templates.py`.
- Unit tests make no network calls (`pytest -m "not db"`). DB tests are marked `@pytest.mark.db`.
- On Windows, async psycopg needs the selector event loop. All entry points use `voice_agent.aio.run()`.
- Before every commit: `uv run ruff format . && uv run ruff check --fix .` (fixes import order), then the task's verify command.
- Commits are SSH-signed (the user's git config). If `git commit` hangs, the signing key isn't loaded: stop and ask the user to run `ssh-add`. Never pass `--no-gpg-sign`.

```python
"""Shared Pydantic models passed between components (architecture §4)."""

from datetime import date
from typing import Any

from pydantic import BaseModel, ConfigDict, Field

from voice_agent.domain.enums import AliasKind, BodyType, City, Fuel, Purpose, SellerType

HARD_FIELDS: tuple[str, ...] = (
    "budget_min", "budget_max", "body_type", "fuel", "city",
    "make", "model", "min_payload_kg", "max_km", "min_year",
)
_MONEY = {"ge": 10_000, "le": 100_000_000}


class Slots(BaseModel):
    """What the buyer has told us so far; shown on screen every turn."""

    model_config = ConfigDict(frozen=True)

    budget_min: int | None = Field(default=None, **_MONEY)
    budget_max: int | None = Field(default=None, **_MONEY)
    budget_target: int | None = Field(default=None, **_MONEY)  # D5: "around X", ranking only
    body_type: BodyType | None = None
    fuel: Fuel | None = None
    city: City | None = None
    make: str | None = None
    model: str | None = None  # canonical model or model family prefix ("Ace")
    min_payload_kg: int | None = Field(default=None, ge=50, le=60_000)
    max_km: int | None = Field(default=None, ge=1_000, le=3_000_000)
    min_year: int | None = Field(default=None, ge=2000, le=2026)
    purpose: Purpose | None = None  # B4: soft signal only, never a filter

    def filled(self) -> dict[str, Any]:
        return self.model_dump(mode="json", exclude_none=True)


class Filters(BaseModel):
    """Hard constraints derived from Slots. Every non-null field becomes one WHERE clause."""

    model_config = ConfigDict(frozen=True)

    budget_min: int | None = None
    budget_max: int | None = None
    body_type: BodyType | None = None
    fuel: Fuel | None = None
    city: City | None = None
    make: str | None = None
    model: str | None = None
    min_payload_kg: int | None = None
    max_km: int | None = None
    min_year: int | None = None

    @classmethod
    def from_slots(cls, slots: Slots) -> "Filters":
        return cls(**{name: getattr(slots, name) for name in HARD_FIELDS})

    def as_dict(self) -> dict[str, Any]:
        return self.model_dump(mode="json", exclude_none=True)

    def replace(self, **changes: Any) -> "Filters":
        return Filters.model_validate({**self.model_dump(), **changes})


class Listing(BaseModel):
    """One row of the listings table (data/README.md)."""

    model_config = ConfigDict(frozen=True)

    listing_id: str
    model_id: str
    make: str
    model: str
    variant: str
    body_type: BodyType
    fuel: Fuel
    year: int
    price_inr: int
    km_driven: int
    payload_kg: int
    gvw_kg: int
    city: City
    registration_state: str
    owners: int
    seller_type: SellerType
    papers_verified: bool
    listing_date: date
    description: str


class ScoredListing(BaseModel):
    """A ranked result with the per-feature data behind its score (E2, Q&A #2)."""

    listing: Listing
    rank: int
    score: float
    features: dict[str, float]
    contributions: dict[str, float]

    def top_reasons(self, n: int = 2) -> list[str]:
        positive = [(v, k) for k, v in self.contributions.items() if v > 0]
        return [k for _, k in sorted(positive, key=lambda p: (-p[0], p[1]))[:n]]


class ResultRef(BaseModel):
    """What conversation memory keeps about a shown result (for 'uska' / 'the second one')."""

    listing_id: str
    rank: int
    make: str
    model: str


class Relaxation(BaseModel):
    """A suggested one-slot loosening that returns at least one row (E3)."""

    slot: str
    old: Any
    new: Any
    count: int


class LowConfidence(BaseModel):
    slot: str
    raw: str
    candidate: str
    score: float


class Unsupported(BaseModel):
    kind: AliasKind
    raw: str
```

```python
"""Protocols for swappable components (C5) and shared provider types."""

from dataclasses import dataclass
from typing import Any, Generic, Literal, Protocol, TypeVar

from pydantic import BaseModel

from voice_agent.domain.enums import AliasKind
from voice_agent.domain.models import Filters, Listing

T = TypeVar("T", bound=BaseModel)


class ProviderError(Exception):
    """A vendor call failed after the provider's own retries."""

    def __init__(self, provider: str, message: str, retryable: bool = False) -> None:
        super().__init__(f"{provider}: {message}")
        self.provider = provider
        self.retryable = retryable


class Usage(BaseModel):
    provider: str
    input_tokens: int = 0
    output_tokens: int = 0


@dataclass(frozen=True)
class LLMResult(Generic[T]):
    value: T
    usage: Usage


class LLM(Protocol):
    name: str

    async def structured(self, system: str, user: str, schema: type[T]) -> LLMResult[T]: ...


class Match(BaseModel):
    raw: str
    canonical: str | None
    score: float
    method: Literal["exact", "trigram", "none"]


class Normalizer(Protocol):
    async def normalize(self, kind: AliasKind, raw: str) -> Match: ...


class SearchResult(BaseModel):
    rows: list[Listing]
    sql: str
    params: dict[str, Any]


class Catalog(Protocol):
    async def search(self, filters: Filters, limit: int = 500) -> SearchResult: ...

    async def count(self, filters: Filters) -> int: ...

    async def get(self, listing_id: str) -> Listing | None: ...
```

```python
    "dedh": 1.5, "dhai": 2.5, "dhaai": 2.5, "adhai": 2.5,
}
_MULT: dict[str, int] = {
    "lakh": 100_000, "lakhs": 100_000, "lac": 100_000, "lacs": 100_000, "l": 100_000,
    "k": 1_000, "hazaar": 1_000, "hazar": 1_000, "thousand": 1_000,
    "crore": 10_000_000, "crores": 10_000_000, "cr": 10_000_000,
}
_NUM = r"\d+(?:\.\d+)?"
_MONEY_UNIT = r"(?:lakhs?|lacs?|l|k|hazaar|hazar|thousand|crores?|cr)"

# Number words become digits only when a unit follows ("do lakh" yes, "do you" no).
_NUM_WORD_RE = re.compile(
    r"\b(" + "|".join(sorted(_WORD_NUM, key=len, reverse=True)) + r")\b"
    r"(?=\s*(?:lakhs?|lacs?|hazaar|hazar|thousand|crores?|tons?|tonnes?|km))"
)
_SAADHE_RE = re.compile(r"\bsa+dhe\s+(\d+(?:\.\d+)?)")
_YEAR_RE = re.compile(
    r"(?:after|since|from|newer than)\s+(20\d\d)\b"
    r"|\b(20\d\d)\s*(?:ke baad|ke bad|se naya|se nayi|or newer|or later|onwards|\+)"
)
_KM_RE = re.compile(rf"({_NUM})\s*(k|hazaar|hazar|thousand|lakhs?)?\s*(?:kms?|kilomet\w*)\b")
_PAYLOAD_RE = re.compile(rf"({_NUM})\s*(tons?|tonnes?|kgs?)\b")
_RANGE_RE = re.compile(
    rf"(?:rs\.?\s*)?({_NUM})\s*({_MONEY_UNIT})?\s*(?:-|to|se|and)\s*"
    rf"(?:rs\.?\s*)?({_NUM})\s*({_MONEY_UNIT})\b"
)
_AMOUNT_RE = re.compile(rf"(?:rs\.?\s*)?({_NUM})\s*({_MONEY_UNIT})\b|rs\.?\s*(\d{{4,}})\b")

_MAX_BEFORE = re.compile(
    r"\b(under|below|within|upto|up to|max|maximum|less than|not more than|at most)\b\D*$"
)
_MAX_AFTER = re.compile(r"^\s*(ke andar|ke under|andar|tak|se kam|or less|max|ke neeche)\b")
_MIN_BEFORE = re.compile(r"\b(above|over|more than|at least|minimum|min)\b\D*$")
_MIN_AFTER = re.compile(r"^\s*(se zyada|se jyada|se upar|or more|plus)\b")
_AROUND_BEFORE = re.compile(r"\b(around|about|approx|approximately|roughly|lagbhag|kareeb)\b\D*$")
_AROUND_AFTER = re.compile(r"^\s*(ke aas ?paas|ke around|ke kareeb)\b")


@dataclass(frozen=True)
class NumericSlots:
    budget_min: int | None = None
    budget_max: int | None = None
    budget_target: int | None = None
    max_km: int | None = None
    min_payload_kg: int | None = None
    min_year: int | None = None

    @property
    def has_budget(self) -> bool:
        return any(v is not None for v in (self.budget_min, self.budget_max, self.budget_target))


def parse_numeric(text: str) -> NumericSlots:
    t = _prepare(text)
    # Order matters: remove year/km/payload spans so they are not read as money.
    min_year, t = _take_year(t)
    max_km, t = _take_km(t)
    min_payload_kg, t = _take_payload(t)
    budget_min, budget_max, budget_target = _take_budget(t)
    return NumericSlots(budget_min, budget_max, budget_target, max_km, min_payload_kg, min_year)


def _prepare(text: str) -> str:
    t = text.lower().replace("₹", " rs ")
    t = re.sub(r"(?<=\d),(?=\d)", "", t)
    t = _NUM_WORD_RE.sub(lambda m: _fmt(_WORD_NUM[m.group(1)]), t)
    return _SAADHE_RE.sub(lambda m: _fmt(float(m.group(1)) + 0.5), t)


def _fmt(x: float) -> str:
    return str(int(x)) if x == int(x) else str(x)


def _blank(t: str, m: re.Match[str]) -> str:
    return t[: m.start()] + " " * (m.end() - m.start()) + t[m.end() :]


def _take_year(t: str) -> tuple[int | None, str]:
    m = _YEAR_RE.search(t)
    if m is None:
        return None, t
    return int(m.group(1) or m.group(2)), _blank(t, m)


def _take_km(t: str) -> tuple[int | None, str]:
    m = _KM_RE.search(t)
    if m is None:
        return None, t
    return round(float(m.group(1)) * _MULT.get(m.group(2) or "", 1)), _blank(t, m)


def _take_payload(t: str) -> tuple[int | None, str]:
    m = _PAYLOAD_RE.search(t)
    if m is None:
        return None, t
    per_unit = 1 if m.group(2).startswith("kg") else 1_000
    return round(float(m.group(1)) * per_unit), _blank(t, m)


def _amount(num: str, unit: str) -> int:
    return round(float(num) * _MULT[unit])


def _take_budget(t: str) -> tuple[int | None, int | None, int | None]:
    m = _RANGE_RE.search(t)
    if m is not None:
        unit_b = m.group(4)
        low, high = _amount(m.group(1), m.group(2) or unit_b), _amount(m.group(3), unit_b)
        return min(low, high), max(low, high), None
    m = _AMOUNT_RE.search(t)
    if m is None:
        return None, None, None
    value = _amount(m.group(1), m.group(2)) if m.group(1) else int(m.group(3))
    before, after = t[max(0, m.start() - 40) : m.start()], t[m.end() : m.end() + 25]
    if _AROUND_BEFORE.search(before) or _AROUND_AFTER.search(after):
        return None, round(value * 1.1), value  # D5: "around X" caps at 1.1·X
    if _MIN_BEFORE.search(before) or _MIN_AFTER.search(after):
        return value, None, None
    return None, value, None  # under/tak/bare amount → max (Assumption)
```

```python
"""Number formatting for speech (F4). The only place display strings for numbers are made."""


def group_indian(n: int) -> str:
    """1,25,000-style grouping."""
    s = str(n)
    if len(s) <= 3:
        return s
    head, tail = s[:-3], s[-3:]
    parts: list[str] = []
    while len(head) > 2:
        parts.insert(0, head[-2:])
        head = head[:-2]
    if head:
        parts.insert(0, head)
    return ",".join([*parts, tail])


def format_price(amount: int) -> str:
    # Listing prices are multiples of 5,000, so two decimals of a lakh are exact.
    if amount >= 100_000 and amount % 1_000 == 0:
        lakhs = f"{amount / 100_000:.2f}".rstrip("0").rstrip(".")
        return f"₹{lakhs} lakh"
    return f"₹{group_indian(amount)}"


def format_km(km: int) -> str:
    return f"{group_indian(km)} km"


def format_kg(kg: int) -> str:
    return f"{group_indian(kg)} kg"
```

```python
"""Slots-derived Filters → parameterized SQL (D1, E1). The only place WHERE clauses are built."""

from typing import Any

from voice_agent.domain.models import Filters, Listing

LISTING_COLUMNS: tuple[str, ...] = (
    "listing_id", "model_id", "make", "model", "variant", "body_type", "fuel", "year",
    "price_inr", "km_driven", "payload_kg", "gvw_kg", "city", "registration_state",
    "owners", "seller_type", "papers_verified", "listing_date", "description",
)
_SELECT = "SELECT " + ", ".join(LISTING_COLUMNS) + " FROM listings"


def _where(f: Filters) -> tuple[str, dict[str, Any]]:
    clauses: list[str] = []
    params: dict[str, Any] = {}

    def add(sql: str, **values: Any) -> None:
        clauses.append(sql)
        params.update(values)

    if f.budget_min is not None:
        add("price_inr >= %(budget_min)s", budget_min=f.budget_min)
    if f.budget_max is not None:
        add("price_inr <= %(budget_max)s", budget_max=f.budget_max)
    if f.body_type is not None:
        add("body_type = %(body_type)s", body_type=f.body_type.value)
    if f.fuel is not None:
        add("fuel = %(fuel)s", fuel=f.fuel.value)
    if f.city is not None:
        add("city = %(city)s", city=f.city.value)
    if f.make is not None:
        add("make = %(make)s", make=f.make)
    if f.model is not None:
        # A family alias like "Ace" covers "Ace Gold", "Ace Mega", ... (stated, not narrowed).
        add("(model = %(model)s OR model LIKE %(model_prefix)s)",
            model=f.model, model_prefix=f"{f.model} %")
    if f.min_payload_kg is not None:
        add("payload_kg >= %(min_payload_kg)s", min_payload_kg=f.min_payload_kg)
    if f.max_km is not None:
        add("km_driven <= %(max_km)s", max_km=f.max_km)
    if f.min_year is not None:
        add("year >= %(min_year)s", min_year=f.min_year)
    where = (" WHERE " + " AND ".join(clauses)) if clauses else ""
    return where, params


def build_search_sql(f: Filters, limit: int = 500) -> tuple[str, dict[str, Any]]:
    where, params = _where(f)
    return f"{_SELECT}{where} ORDER BY listing_id LIMIT %(limit)s", {**params, "limit": limit}


def build_count_sql(f: Filters) -> tuple[str, dict[str, Any]]:
    where, params = _where(f)
    return f"SELECT count(*) FROM listings{where}", params


def constraint_violations(listing: Listing, f: Filters) -> list[str]:
    """F2.3: independent Python re-check of every hard filter (defense in depth vs SQL bugs)."""
    checks: list[tuple[str, bool]] = [
        ("budget_min", f.budget_min is not None and listing.price_inr < f.budget_min),
        ("budget_max", f.budget_max is not None and listing.price_inr > f.budget_max),
        ("body_type", f.body_type is not None and listing.body_type != f.body_type),
        ("fuel", f.fuel is not None and listing.fuel != f.fuel),
        ("city", f.city is not None and listing.city != f.city),
        ("make", f.make is not None and listing.make != f.make),
        ("model", f.model is not None
         and not (listing.model == f.model or listing.model.startswith(f"{f.model} "))),
        ("min_payload_kg", f.min_payload_kg is not None and listing.payload_kg < f.min_payload_kg),
        ("max_km", f.max_km is not None and listing.km_driven > f.max_km),
        ("min_year", f.min_year is not None and listing.year < f.min_year),
    ]
    return [name for name, violated in checks if violated]
```

```python
"""Transparent weighted ranking (E2): every score is a sum of visible per-feature contributions."""

from collections.abc import Mapping, Sequence
from pathlib import Path

import yaml

from voice_agent.domain.enums import BodyType, Fuel, Purpose
from voice_agent.domain.models import Listing, ScoredListing, Slots

DEFAULT_WEIGHTS: dict[str, float] = {
    "price": 0.30, "low_km": 0.20, "newer_year": 0.20, "papers_verified": 0.15, "purpose_fit": 0.15,
}
_SMALL_BODIES = {BodyType.THREE_WHEELER, BodyType.MINI_TRUCK, BodyType.PICKUP, BodyType.CARGO_VAN}
_CITY_FUEL = {Fuel.CNG: 1.0, Fuel.ELECTRIC: 1.0, Fuel.PETROL: 0.6, Fuel.DIESEL: 0.4}


def load_weights(path: Path) -> dict[str, float]:
    raw = yaml.safe_load(path.read_text()) or {}
    unknown = set(raw) - set(DEFAULT_WEIGHTS)
    if unknown:
        raise ValueError(f"unknown ranking features: {sorted(unknown)}")
    return {name: float(value) for name, value in raw.items()}


def rank(
    rows: Sequence[Listing], slots: Slots, weights: Mapping[str, float], top_k: int = 3
) -> list[ScoredListing]:
    if not rows:
        return []
    prices = [r.price_inr for r in rows]
    kms = [r.km_driven for r in rows]
    years = [r.year for r in rows]
    scored: list[tuple[float, str, Listing, dict[str, float], dict[str, float]]] = []
    for r in rows:
        features = {
            "price": _price_feature(r.price_inr, prices, slots.budget_target),
            "low_km": _inverted(r.km_driven, kms),
            "newer_year": _scaled(r.year, years),
            "papers_verified": 1.0 if r.papers_verified else 0.0,
            "purpose_fit": purpose_fit(r, slots.purpose),
        }
        contributions = {k: round(weights.get(k, 0.0) * v, 4) for k, v in features.items()}
        scored.append((round(sum(contributions.values()), 4), r.listing_id, r, features, contributions))
    scored.sort(key=lambda s: (-s[0], s[1]))
    return [
        ScoredListing(listing=r, rank=i + 1, score=score, features=f, contributions=c)
        for i, (score, _, r, f, c) in enumerate(scored[:top_k])
    ]


def purpose_fit(listing: Listing, purpose: Purpose | None) -> float:
    if purpose is None:
        return 0.0
    if purpose is Purpose.CITY_DELIVERY:
        size = 1.0 if listing.body_type in _SMALL_BODIES else 0.3
        return round((_CITY_FUEL[listing.fuel] + size) / 2, 4)
    if purpose is Purpose.LONG_HAUL:
        fuel = 1.0 if listing.fuel is Fuel.DIESEL else 0.2
        size = 1.0 if listing.payload_kg >= 3_000 else 0.5 if listing.payload_kg >= 1_000 else 0.0
        return round((fuel + size) / 2, 4)
    if purpose is Purpose.CONSTRUCTION:
        return {BodyType.TIPPER: 1.0, BodyType.ICV: 0.6, BodyType.LCV: 0.6}.get(listing.body_type, 0.2)
    return {BodyType.PICKUP: 1.0, BodyType.MINI_TRUCK: 0.8, BodyType.THREE_WHEELER: 0.6}.get(
        listing.body_type, 0.4
    )


def _scaled(value: int, values: Sequence[int]) -> float:
    lo, hi = min(values), max(values)
    return 1.0 if hi == lo else round((value - lo) / (hi - lo), 4)


def _inverted(value: int, values: Sequence[int]) -> float:
    lo, hi = min(values), max(values)
    return 1.0 if hi == lo else round((hi - value) / (hi - lo), 4)


def _price_feature(price: int, prices: Sequence[int], target: int | None) -> float:
    if target is None:
        return _inverted(price, prices)
    return round(max(0.0, 1.0 - abs(price - target) / (0.1 * target)), 4)
```

```python
"""Zero-result relaxation (E3): find the smallest one-slot loosening with ≥1 row. Never auto-applied."""

import math
from collections.abc import Awaitable, Callable, Mapping, Sequence
from typing import Any

from voice_agent.domain.enums import BodyType, City
from voice_agent.domain.models import Filters, Relaxation

CountFn = Callable[[Filters], Awaitable[int]]

ADJACENT_BODY: dict[BodyType, BodyType] = {
    BodyType.MINI_TRUCK: BodyType.PICKUP,
    BodyType.PICKUP: BodyType.MINI_TRUCK,
    BodyType.THREE_WHEELER: BodyType.MINI_TRUCK,
    BodyType.CARGO_VAN: BodyType.MINI_TRUCK,
    BodyType.LCV: BodyType.ICV,
    BodyType.ICV: BodyType.LCV,
    BodyType.TIPPER: BodyType.ICV,
}
_DROPPABLE = ("min_year", "max_km", "min_payload_kg", "model", "make", "budget_min")


def candidate_changes(f: Filters, neighbors: Mapping[City, Sequence[City]]) -> list[tuple[str, Any]]:
    """Ordered smallest-change first: +20% budget, nearby city, any fuel, adjacent body, drop extras."""
    changes: list[tuple[str, Any]] = []
    if f.budget_max is not None:
        changes.append(("budget_max", _round_up(f.budget_max * 1.2, 10_000)))
    if f.city is not None:
        changes.extend(("city", n) for n in neighbors.get(f.city, []))
    if f.fuel is not None:
        changes.append(("fuel", None))
    if f.body_type is not None:
        changes.append(("body_type", ADJACENT_BODY[f.body_type]))
    changes.extend((name, None) for name in _DROPPABLE if getattr(f, name) is not None)
    return changes


async def propose_relaxation(
    f: Filters, count: CountFn, neighbors: Mapping[City, Sequence[City]]
) -> Relaxation | None:
    for slot, new in candidate_changes(f, neighbors):
        n = await count(f.replace(**{slot: new}))
        if n > 0:
            return Relaxation(slot=slot, old=getattr(f, slot), new=new, count=n)
    return None


def _round_up(value: float, step: int) -> int:
    return int(math.ceil(value / step) * step)
```

```python
from collections.abc import Sequence

from voice_agent.domain.models import ResultRef
from voice_agent.understanding.schema import Reference


def resolve_reference(
    ref: Reference | None, last: Sequence[ResultRef], focused: str | None
) -> str | None:
    if not last:
        return None
    if ref is not None and ref.ordinal is not None:
        idx = len(last) - 1 if ref.ordinal == -1 else ref.ordinal - 1
        return last[idx].listing_id if 0 <= idx < len(last) else None
    if ref is not None and (ref.make or ref.model):
        matches = [r for r in last if _matches(r, ref)]
        return matches[0].listing_id if len(matches) == 1 else None  # ambiguous → ask
    return focused or last[0].listing_id


def _matches(r: ResultRef, ref: Reference) -> bool:
    if ref.make and ref.make.lower() not in r.make.lower():
        return False
    return not (ref.model and ref.model.lower() not in r.model.lower())
```

```python
"""Pure conversation-state transitions (G1–G3, D6, E3). No I/O."""

from collections.abc import Mapping
from enum import Enum
from typing import Any

from pydantic import ValidationError

from voice_agent.domain.enums import FollowUpField, Intent
from voice_agent.domain.models import Slots
from voice_agent.state.models import Memory, PendingConfirmation, Route, SlotDiff
from voice_agent.state.references import resolve_reference
from voice_agent.understanding.schema import Understanding


def sanitize_updates(updates: Mapping[str, Any]) -> tuple[dict[str, Any], list[str]]:
    """D7: drop unknown keys and out-of-range/invalid values instead of passing them on."""
    valid: dict[str, Any] = {}
    dropped: list[str] = []
    for key, value in updates.items():
        if key not in Slots.model_fields:
            dropped.append(key)
            continue
        try:
            Slots.model_validate({key: value})
        except ValidationError:
            dropped.append(key)
            continue
        valid[key] = value
    return valid, dropped


def merge_slots(old: Slots, updates: Mapping[str, Any]) -> tuple[Slots, SlotDiff]:
    new = Slots.model_validate({**old.model_dump(), **updates})
    diff = {
        name: (_plain(getattr(old, name)), _plain(getattr(new, name)))
        for name in Slots.model_fields
        if getattr(old, name) != getattr(new, name)
    }
    return new, diff


def apply_understanding(mem: Memory, u: Understanding) -> tuple[Memory, SlotDiff, Route]:
    ti = u.turn_intent
    if u.unsupported:
        return mem, {}, Route(kind="unsupported", unsupported=u.unsupported)

    if mem.pending_confirmation is not None and ti.confirmation is not None:
        pending = mem.pending_confirmation
        updates = dict(u.updates)
        if ti.confirmation == "yes":
            updates = {**pending.deferred_updates, pending.slot: pending.candidate, **updates}
        return _search_with(mem, updates)

    if u.low_confidence:
        lc = u.low_confidence[0]
        pending = PendingConfirmation(
            slot=lc.slot, raw=lc.raw, candidate=lc.candidate, score=lc.score,
            deferred_updates=u.updates,
        )
        return mem.model_copy(update={"pending_confirmation": pending}), {}, Route(kind="confirm")

    if ti.intent is Intent.RESET:
        diff = {k: (v, None) for k, v in mem.slots.filled().items()}
        return Memory(), diff, Route(kind="reset")

    if ti.intent is Intent.OUT_OF_DOMAIN:
        return mem, {}, Route(kind="out_of_domain")

    if ti.intent is Intent.FOLLOW_UP:
        listing_id = resolve_reference(ti.reference, mem.last_results, mem.focused_listing_id)
        if listing_id is None:
            return mem, {}, Route(kind="which_one")
        return (
            mem.model_copy(update={"focused_listing_id": listing_id}),
            {},
            Route(kind="follow_up", listing_id=listing_id,
                  follow_up_field=ti.follow_up_field or FollowUpField.SUMMARY),
        )

    if ti.intent is Intent.RELAX and mem.pending_relaxation is not None:
        r = mem.pending_relaxation
        return _search_with(mem, {r.slot: r.new})

    return _search_with(mem, u.updates)


def _search_with(mem: Memory, updates: Mapping[str, Any]) -> tuple[Memory, SlotDiff, Route]:
    slots, diff = merge_slots(mem.slots, updates)
    new_mem = mem.model_copy(
        update={"slots": slots, "pending_confirmation": None, "pending_relaxation": None}
    )
    if not slots.filled():
        return new_mem, diff, Route(kind="need_info")
    return new_mem, diff, Route(kind="search")


def _plain(value: Any) -> Any:
    return value.value if isinstance(value, Enum) else value
```

```python
"""Structured response shape (architecture §9): per-vehicle sentences bound to listing ids."""

from typing import Literal

from pydantic import BaseModel, Field


class ItemSentence(BaseModel):
    listing_id: str
    text: str


class ComposedResponse(BaseModel):
    intro: str = Field(default="", description="Opening sentence. No numbers, no vehicle names.")
    items: list[ItemSentence] = Field(default_factory=list, description="One sentence per vehicle")
    closing: str = Field(default="", description="Closing question. No numbers, no vehicle names.")

    def sentences(self) -> list[str]:
        parts = [self.intro, *(i.text for i in self.items), self.closing]
        return [p.strip() for p in parts if p.strip()]

    def text(self) -> str:
        return " ".join(self.sentences())


class Violation(BaseModel):
    where: str
    kind: Literal["order", "number", "number_word", "entity", "extra_vehicle"]
    detail: str


class Verdict(BaseModel):
    ok: bool
    violations: list[Violation] = Field(default_factory=list)
    source: Literal["llm", "template"]
```

```python
"""Fact sheet: the only data the response LLM sees, with numbers pre-formatted in code (F1, F4)."""

import re
from collections.abc import Sequence

from pydantic import BaseModel

from voice_agent.domain.enums import BodyType, Language, Purpose, SellerType
from voice_agent.domain.models import Listing, ScoredListing
from voice_agent.response.formatting import format_kg, format_km, format_price

NUM_TOKEN = re.compile(r"\d[\d,]*(?:\.\d+)?")
BODY_LABEL: dict[BodyType, str] = {
    BodyType.THREE_WHEELER: "three-wheeler", BodyType.MINI_TRUCK: "mini truck",
    BodyType.PICKUP: "pickup", BodyType.CARGO_VAN: "cargo van", BodyType.LCV: "light truck",
    BodyType.ICV: "medium truck", BodyType.TIPPER: "tipper",
}
SELLER_LABEL = {SellerType.DEALER: "a dealer", SellerType.INDIVIDUAL: "an individual seller"}
PURPOSE_REASON: dict[Purpose, str] = {
    Purpose.CITY_DELIVERY: "suits city delivery",
    Purpose.LONG_HAUL: "suits long-distance runs",
    Purpose.CONSTRUCTION: "built for construction work",
    Purpose.AGRICULTURE: "handy for farm use",
}


def numeric_tokens(text: str) -> set[str]:
    return {t.replace(",", "") for t in NUM_TOKEN.findall(text)}


class FactItem(BaseModel):
    listing_id: str
    rank: int
    title: str
    make: str
    model: str
    city: str
    price: str
    km: str
    payload: str
    gvw: str
    year: str
    fuel: str
    body_type: str
    owners: str
    papers: str
    seller: str
    registration: str
    reasons: list[str]

    def allowed_numbers(self) -> set[str]:
        fields = [self.title, self.price, self.km, self.payload, self.gvw, self.year, self.owners]
        return numeric_tokens(" ".join([*fields, *self.reasons]))


class FactSheet(BaseModel):
    language: Language
    items: list[FactItem]


def build_fact_item(
    listing: Listing, rank: int, reason_features: Sequence[str], purpose: Purpose | None
) -> FactItem:
    item = FactItem(
        listing_id=listing.listing_id,
        rank=rank,
        title=f"{listing.year} {listing.make} {listing.model} {listing.variant}",
        make=listing.make,
        model=listing.model,
        city=listing.city.value.title(),
        price=format_price(listing.price_inr),
        km=format_km(listing.km_driven),
        payload=format_kg(listing.payload_kg),
        gvw=format_kg(listing.gvw_kg),
        year=str(listing.year),
        fuel="CNG" if listing.fuel.value == "cng" else listing.fuel.value,
        body_type=BODY_LABEL[listing.body_type],
        owners=f"{listing.owners} owner" + ("" if listing.owners == 1 else "s"),
        papers="papers verified" if listing.papers_verified else "papers not verified",
        seller=SELLER_LABEL[listing.seller_type],
        registration=listing.registration_state,
        reasons=[],
    )
    reasons = [_reason(f, item, purpose) for f in reason_features]
    return item.model_copy(update={"reasons": [r for r in reasons if r]})


def build_fact_sheet(
    results: Sequence[ScoredListing], purpose: Purpose | None, language: Language
) -> FactSheet:
    return FactSheet(
        language=language,
        items=[build_fact_item(s.listing, s.rank, s.top_reasons(), purpose) for s in results],
    )


def _reason(feature: str, item: FactItem, purpose: Purpose | None) -> str:
    if feature == "price":
        return f"priced at {item.price}"
    if feature == "low_km":
        return f"only {item.km} driven"
    if feature == "newer_year":
        return f"a {item.year} model"
    if feature == "papers_verified":
        return "papers verified"
    if feature == "purpose_fit" and purpose is not None:
        return PURPOSE_REASON[purpose]
    return ""
```

- [ ] **Step 5: Write `src/voice_agent/response/validator.py`**

```python
"""Code-level anti-hallucination check (F2) for LLM-composed responses. Pure, no I/O.

Known limits (F3): subjective claims without numbers or entities, coincidentally equal numbers,
unsupported comparatives ("cheapest in Pune"), and "one"/"do" are not treated as number words.
"""

import re
from dataclasses import dataclass

from voice_agent.response.factsheet import FactSheet, numeric_tokens
from voice_agent.response.schema import ComposedResponse, Verdict, Violation

_NUMBER_WORDS = re.compile(
    r"\b(two|three|four|five|six|seven|eight|nine|ten|eleven|twelve|fifteen|twenty|thirty|"
    r"forty|fifty|hundred|thousand|ek|chaar|paanch|chhe|saat|aath|nau|dedh|dhai|hazaar|sau)\b"
)


@dataclass(frozen=True)
class EntityVocab:
    makes: frozenset[str]
    models: frozenset[str]

    def find(self, text: str) -> list[tuple[str, str]]:
        """(kind, canonical) for every make/model named in text; longest names matched first."""
        found: list[tuple[str, str]] = []
        lowered = text.lower()
        for kind, names in (("model", self.models), ("make", self.makes)):
            for name in sorted(names, key=len, reverse=True):
                pattern = r"(?<![\w])" + re.escape(name.lower()) + r"(?![\w])"
                if re.search(pattern, lowered):
                    found.append((kind, name))
                    lowered = re.sub(pattern, " ", lowered)
        return found


def validate(resp: ComposedResponse, facts: FactSheet, vocab: EntityVocab) -> Verdict:
    violations: list[Violation] = []
    expected = [i.listing_id for i in facts.items]
    got = [i.listing_id for i in resp.items]
    if got != expected:
        violations.append(Violation(where="items", kind="order", detail=f"expected {expected}, got {got}"))

    by_id = {i.listing_id: i for i in facts.items}
    for sentence in resp.items:
        fact = by_id.get(sentence.listing_id)
        if fact is None:
            continue
        violations += _numbers(sentence.text, fact.allowed_numbers(), sentence.listing_id)
        for kind, name in vocab.find(sentence.text):
            own = fact.make if kind == "make" else fact.model
            if name != own:
                violations.append(Violation(where=sentence.listing_id, kind="entity",
                                            detail=f"{kind} '{name}' is not this listing's"))

    for where, text in (("intro", resp.intro), ("closing", resp.closing)):
        violations += _numbers(text, set(), where)
        violations += [Violation(where=where, kind="extra_vehicle", detail=name)
                       for _, name in vocab.find(text)]
    return Verdict(ok=not violations, violations=violations, source="llm")


def _numbers(text: str, allowed: set[str], where: str) -> list[Violation]:
    out = [Violation(where=where, kind="number", detail=t)
           for t in sorted(numeric_tokens(text) - allowed)]
    out += [Violation(where=where, kind="number_word", detail=m.group(0))
            for m in _NUMBER_WORDS.finditer(text.lower())]
    return out
```

```python
def results(facts: FactSheet) -> ComposedResponse:

def describe_slots(slots: Slots) -> str:
    noun = BODY_LABEL[slots.body_type] if slots.body_type else "vehicle"
    parts = []
    if slots.fuel:
        parts.append("CNG" if slots.fuel is Fuel.CNG else slots.fuel.value)
    if slots.make:
        parts.append(slots.make)
    if slots.model:
        parts.append(slots.model)
    parts.append(noun)
    if slots.city:
        parts.append(f"in {slots.city.value.title()}")
    if slots.budget_min and slots.budget_max:
        parts.append(f"between {format_price(slots.budget_min)} and {format_price(slots.budget_max)}")
    elif slots.budget_max:
        parts.append(f"under {format_price(slots.budget_max)}")
    elif slots.budget_min:
        parts.append(f"above {format_price(slots.budget_min)}")
    if slots.max_km:
        parts.append(f"under {format_km(slots.max_km)}")
    if slots.min_year:
        parts.append(f"from {slots.min_year} or newer")
    return " ".join(parts)


def zero_results(slots: Slots, relaxation: Relaxation | None) -> ComposedResponse:
    intro = f"I couldn't find any {describe_slots(slots)}."
    closing = _offer(relaxation) if relaxation else "Try changing the budget, city or fuel."
    return ComposedResponse(intro=intro, closing=closing)


def _offer(r: Relaxation) -> str:
    n = r.count
    are, s = ("is", "") if n == 1 else ("are", "s")
    if r.slot == "budget_max":
        return f"If you can go up to {format_price(int(r.new))}, there {are} {n} option{s}. Shall I show them?"
    if r.slot == "city":
        return f"There {are} {n} in {City(r.new).value.title()}. Want to see those?"
    if r.slot == "fuel":
        return f"With any fuel, there {are} {n}. Shall I include all fuels?"
    if r.slot == "body_type":
        return f"I found {n} {BODY_LABEL[BodyType(r.new)]} option{s} instead. Want to see them?"
    return f"Without the {SLOT_LABEL.get(r.slot, r.slot)} limit, there {are} {n}. Shall I drop it?"

    return _say(f"Just to check, did you mean {pending.candidate}?")


def unsupported(items: Sequence[Unsupported]) -> ComposedResponse:
    first = items[0]
    if first.kind is AliasKind.CITY:
        cities = ", ".join(c.value.title() for c in City)
        return _say(f"Sorry, I don't have listings in {first.raw.title()}. I cover {cities}.")
    return _say(f"Sorry, I couldn't find {first.raw} in the catalog.")


def follow_up(item: FactItem, field: FollowUpField) -> ComposedResponse:
    answers = {
        FollowUpField.PRICE: f"The {item.title} is priced at {item.price}.",
        FollowUpField.KM: f"It has done {item.km}.",
        FollowUpField.PAYLOAD: f"Its payload is {item.payload}.",
        FollowUpField.GVW: f"Its gross vehicle weight is {item.gvw}.",
        FollowUpField.YEAR: f"It's a {item.year} model.",
        FollowUpField.FUEL: f"It runs on {item.fuel}.",
        FollowUpField.OWNERS: f"It has had {item.owners}.",
        FollowUpField.PAPERS: f"Its {item.papers}." if item.papers.startswith("papers verified")

        FollowUpField.SELLER: f"It's sold by {item.seller}.",

                                f"{item.payload} payload, {item.owners}, {item.papers}."),
    }
    text = answers.get(field)
    if text is None:
        text = ("That detail isn't in the listing. I can tell you the price, km, payload, year, "
                "fuel, owners, papers or location.")
    return ComposedResponse(items=[ItemSentence(listing_id=item.listing_id, text=text)])
```

```python
"""Create tables and load data/*.csv on first boot (idempotent)."""

import csv
from pathlib import Path

from psycopg_pool import AsyncConnectionPool

from voice_agent.aio import run
from voice_agent.catalog.filters import LISTING_COLUMNS
from voice_agent.config import get_settings
from voice_agent.domain.enums import BodyType, City, Fuel
from voice_agent.domain.models import Listing

SCHEMA_PATH = Path(__file__).with_name("schema.sql")
_INSERT_LISTING = (
    f"INSERT INTO listings ({', '.join(LISTING_COLUMNS)}) VALUES "
    f"({', '.join(f'%({c})s' for c in LISTING_COLUMNS)})"
)
_INSERT_ALIAS = (
    "INSERT INTO aliases (alias, canonical, kind) VALUES (%(alias)s, %(canonical)s, %(kind)s) "
    "ON CONFLICT DO NOTHING"
)
_INSERT_NEIGHBOR = (
    "INSERT INTO city_neighbors (city, neighbor, distance_km) "
    "VALUES (%(city)s, %(neighbor)s, %(distance_km)s) ON CONFLICT DO NOTHING"
)


def _read(path: Path) -> list[dict[str, str]]:
    with path.open(newline="", encoding="utf-8") as fh:
        return list(csv.DictReader(fh))


def _self_aliases(listings: list[Listing]) -> list[dict[str, str]]:
    """Canonical names are aliases of themselves, so exact matches need no special case."""
    rows = [{"alias": x.model.lower(), "canonical": x.model, "kind": "model"} for x in listings]
    rows += [{"alias": x.make.lower(), "canonical": x.make, "kind": "make"} for x in listings]
    rows += [{"alias": c.value, "canonical": c.value, "kind": "city"} for c in City]
    rows += [{"alias": b.value.replace("_", " "), "canonical": b.value, "kind": "body_type"}
             for b in BodyType]
    rows += [{"alias": f.value, "canonical": f.value, "kind": "fuel"} for f in Fuel]
    return rows


async def seed(pool: AsyncConnectionPool, data_dir: Path) -> None:
    async with pool.connection() as conn:
        await conn.execute(SCHEMA_PATH.read_text(encoding="utf-8"))
        cur = await conn.execute("SELECT count(*) FROM listings")
        row = await cur.fetchone()
        if row and row[0]:
            return
        listings = [Listing.model_validate(r) for r in _read(data_dir / "listings.csv")]
        async with conn.cursor() as c:
            await c.executemany(_INSERT_LISTING, [x.model_dump(mode="json") for x in listings])
            await c.executemany(_INSERT_ALIAS, _read(data_dir / "aliases.csv") + _self_aliases(listings))
            await c.executemany(_INSERT_NEIGHBOR, _read(data_dir / "city_neighbors.csv"))


async def _main() -> None:
    settings = get_settings()
    async with AsyncConnectionPool(settings.database_url, open=False,
                                   kwargs={"autocommit": True}) as pool:
        await seed(pool, settings.data_dir)
    print("seeded")


if __name__ == "__main__":
    run(_main())
```

```python
        sql, params = build_search_sql(filters, limit)
        async with self._pool.connection() as conn, conn.cursor(row_factory=dict_row) as cur:
            await cur.execute(sql, params)
            rows = await cur.fetchall()
        return SearchResult(rows=[Listing.model_validate(r) for r in rows], sql=sql, params=params)

    async def count(self, filters: Filters) -> int:
        sql, params = build_count_sql(filters)
        async with self._pool.connection() as conn:
            cur = await conn.execute(sql, params)
            row = await cur.fetchone()
        return int(row[0]) if row else 0

    async def get(self, listing_id: str) -> Listing | None:
        result = await self._search_by_id(listing_id)
        return result[0] if result else None

    async def _search_by_id(self, listing_id: str) -> list[Listing]:
        async with self._pool.connection() as conn, conn.cursor(row_factory=dict_row) as cur:
            await cur.execute("SELECT * FROM listings WHERE listing_id = %(id)s", {"id": listing_id})
            return [Listing.model_validate(r) for r in await cur.fetchall()]


class PgNormalizer:
    """Raw mention → canonical value: exact alias first, then pg_trgm similarity (D2)."""

    def __init__(self, pool: AsyncConnectionPool) -> None:
        self._pool = pool

    async def normalize(self, kind: AliasKind, raw: str) -> Match:
        q = raw.strip().lower()
        params = {"kind": kind.value, "q": q}
        async with self._pool.connection() as conn:
            cur = await conn.execute(
                "SELECT canonical FROM aliases WHERE kind = %(kind)s AND alias = %(q)s", params)
            row = await cur.fetchone()
            if row:
                return Match(raw=raw, canonical=row[0], score=1.0, method="exact")
            cur = await conn.execute(
                "SELECT canonical, similarity(alias, %(q)s) AS score FROM aliases "
                "WHERE kind = %(kind)s ORDER BY score DESC LIMIT 1", params)
            row = await cur.fetchone()
        if row and float(row[1]) > 0:
            return Match(raw=raw, canonical=row[0], score=round(float(row[1]), 3), method="trigram")
        return Match(raw=raw, canonical=None, score=0.0, method="none")


async def load_alias_map(pool: AsyncConnectionPool) -> dict[AliasKind, dict[str, str]]:
    out: dict[AliasKind, dict[str, str]] = {k: {} for k in AliasKind}
    async with pool.connection() as conn:
        cur = await conn.execute("SELECT alias, canonical, kind FROM aliases")
        for alias, canonical, kind in await cur.fetchall():
            out[AliasKind(kind)][alias] = canonical
    return out


async def load_vocab(pool: AsyncConnectionPool) -> EntityVocab:

        rows = await cur.fetchall()

            "SELECT city, neighbor FROM city_neighbors ORDER BY city, distance_km")

    return out
```

```python
"""LLM providers (I1, I2, I3): LangChain chat models behind the LLM protocol, with failover."""

import logging
from typing import Any, TypeVar

from langchain_core.language_models import BaseChatModel
from pydantic import BaseModel

from voice_agent.config import Settings
from voice_agent.domain.ports import LLM, LLMResult, ProviderError, Usage

T = TypeVar("T", bound=BaseModel)
log = logging.getLogger(__name__)


class LangChainLLM:
    def __init__(self, name: str, chat: BaseChatModel) -> None:
        self.name = name
        self._chat = chat

    async def structured(self, system: str, user: str, schema: type[T]) -> LLMResult[T]:
        runnable = self._chat.with_structured_output(schema, include_raw=True)
        try:
            out: dict[str, Any] = await runnable.ainvoke([("system", system), ("human", user)])  # type: ignore[assignment]
        except Exception as exc:  # vendor SDKs raise many types; normalise at the seam
            raise ProviderError(self.name, str(exc), retryable=True) from exc
        parsed = out.get("parsed")
        if parsed is None:
            raise ProviderError(self.name, f"unparseable output: {out.get('parsing_error')}")
        meta = getattr(out.get("raw"), "usage_metadata", None) or {}
        usage = Usage(provider=self.name, input_tokens=meta.get("input_tokens", 0),
                      output_tokens=meta.get("output_tokens", 0))
        return LLMResult(value=parsed, usage=usage)


class FailoverLLM:
    """Tries the primary; on ProviderError tries the fallback once. Usage.provider shows who served."""

    def __init__(self, primary: LLM, fallback: LLM | None) -> None:
        self.name = primary.name
        self._primary = primary
        self._fallback = fallback

    async def structured(self, system: str, user: str, schema: type[T]) -> LLMResult[T]:
        try:
            return await self._primary.structured(system, user, schema)
        except ProviderError as exc:
            if self._fallback is None:
                raise
            log.warning("llm failover %s -> %s: %s", self._primary.name, self._fallback.name, exc)
            return await self._fallback.structured(system, user, schema)


def _groq(settings: Settings) -> LLM | None:
    if settings.groq_api_key is None:
        return None
    from langchain_groq import ChatGroq

    chat = ChatGroq(model=settings.groq_model, api_key=settings.groq_api_key,
                    temperature=0, max_retries=2, timeout=15)
    return LangChainLLM("groq", chat)


def _gemini(settings: Settings) -> LLM | None:
    if settings.gemini_api_key is None:
        return None
    from langchain_google_genai import ChatGoogleGenerativeAI

    chat = ChatGoogleGenerativeAI(model=settings.gemini_model,
                                  google_api_key=settings.gemini_api_key.get_secret_value(),
                                  temperature=0, max_retries=2, timeout=15)
    return LangChainLLM("gemini", chat)


_FACTORIES = {"groq": _groq, "gemini": _gemini}


def make_llm(settings: Settings) -> LLM:
    primary = _FACTORIES[settings.llm_provider](settings)
    if primary is None:
        raise RuntimeError(f"{settings.llm_provider.upper()}_API_KEY is not set (see .env.example)")
    fallback = None
    if settings.llm_fallback_provider not in ("none", settings.llm_provider):
        fallback = _FACTORIES[settings.llm_fallback_provider](settings)
    return FailoverLLM(primary, fallback)
```

```python
"""Rules-only extractor: the LLM-outage fallback (D3, H5). Exact alias matching, no guessing."""

import re
from collections.abc import Mapping

from voice_agent.domain.enums import AliasKind, BodyType, Fuel, Intent
from voice_agent.understanding.schema import ExtractedSlots, TurnIntent

_RESET = re.compile(r"\b(start over|reset|naya search|shuru se)\b")


def rules_extract(text: str, alias_map: Mapping[AliasKind, Mapping[str, str]]) -> TurnIntent:
    padded = " " + re.sub(r"[^\w\s]", " ", text.lower()) + " "
    padded = re.sub(r"\s+", " ", padded)
    if _RESET.search(padded):
        return TurnIntent(intent=Intent.RESET)
    body = _hit(padded, alias_map.get(AliasKind.BODY_TYPE, {}))
    fuel = _hit(padded, alias_map.get(AliasKind.FUEL, {}))
    city = _hit(padded, alias_map.get(AliasKind.CITY, {}))
    make = _hit(padded, alias_map.get(AliasKind.MAKE, {}))
    model = _hit(padded, alias_map.get(AliasKind.MODEL, {}))
    slots = ExtractedSlots(
        body_type=BodyType(body[1]) if body else None,
        fuel=Fuel(fuel[1]) if fuel else None,
        city_mention=city[0] if city else None,
        make_mention=make[0] if make else None,
        model_mention=model[0] if model else None,
    )
    return TurnIntent(intent=Intent.SEARCH, slots=slots)


def _hit(padded: str, aliases: Mapping[str, str]) -> tuple[str, str] | None:
    for alias in sorted(aliases, key=len, reverse=True):
        if f" {alias} " in padded:
            return alias, aliases[alias]
    return None
```

```python
"""LLM slot extraction via one structured tool call (D1). Prompt vocab comes from the enums (D2)."""

import json
from collections.abc import Mapping

from voice_agent.domain.enums import AliasKind, BodyType, Fuel, Purpose
from voice_agent.domain.ports import LLM, LLMResult
from voice_agent.understanding.schema import ExtractionContext, TurnIntent

_SYSTEM = """You extract search intent for a used commercial-vehicle marketplace in India.
Fill the TurnIntent schema. Rules:
- intent: "search" (new or extra constraints), "correct" (user says a value they gave was wrong,
  e.g. "no, not diesel, CNG"), "follow_up" (question about a vehicle already shown),
  "relax" (user accepts the relaxation the assistant just offered), "reset" (start over),
  "out_of_domain" (anything unrelated to finding a vehicle).
- Only fill slots stated in THIS utterance. Never copy slots from the current state.
- body_type is one of: {body_types}. fuel is one of: {fuels}. purpose is one of: {purposes}.
- Phrase hints from our alias table: {body_hints}; {fuel_hints}.
- city_mention / make_mention / model_mention: copy the words exactly as the user said them.
  Do not correct spelling, translate or guess.
- Money is in rupees as integers (5 lakh = 500000). "under / ke andar / tak X" -> budget_max;
  "around X" -> budget_target; "X to Y" -> budget_min and budget_max.
- clear_slots: slots the user explicitly drops ("any fuel is fine" -> ["fuel"]).
- follow_up: set reference (ordinal 1-3, -1 for the last one; or make/model; or pronoun=true for
  it / uska / that one) and follow_up_field.
- confirmation: "yes" or "no" only if the assistant's pending question is answered.
- language: "hi-en" if the user speaks Hindi or Hinglish, otherwise "en"."""


def build_system_prompt(alias_map: Mapping[AliasKind, Mapping[str, str]]) -> str:
    def hints(kind: AliasKind) -> str:
        pairs = sorted(alias_map.get(kind, {}).items())
        return ", ".join(f"'{a}'={c}" for a, c in pairs if a != c)

    return _SYSTEM.format(
        body_types=", ".join(b.value for b in BodyType),
        fuels=", ".join(f.value for f in Fuel),
        purposes=", ".join(p.value for p in Purpose),
        body_hints=hints(AliasKind.BODY_TYPE),
        fuel_hints=hints(AliasKind.FUEL),
    )


def build_user_prompt(text: str, ctx: ExtractionContext) -> str:
    context = {
        "current_slots": ctx.slots.filled(),
        "last_results": [{"rank": r.rank, "make": r.make, "model": r.model} for r in ctx.last_results],
        "pending_question": ctx.pending_question,
    }
    return f"Context: {json.dumps(context, ensure_ascii=False)}\nUser said: {text}"


class LLMSlotExtractor:
    def __init__(self, llm: LLM, alias_map: Mapping[AliasKind, Mapping[str, str]]) -> None:
        self._llm = llm
        self._system = build_system_prompt(alias_map)

    @property
    def primary_provider(self) -> str:
        return self._llm.name

    async def extract(self, text: str, ctx: ExtractionContext) -> LLMResult[TurnIntent]:
        return await self._llm.structured(self._system, build_user_prompt(text, ctx), TurnIntent)
```

```python
from voice_agent.understanding.numbers import NumericSlots, parse_numeric
from voice_agent.understanding.rules import rules_extract
from voice_agent.understanding.schema import ExtractionContext, TurnIntent, Understanding

TRGM_ACCEPT = 0.6   # D6: at or above → accept silently
TRGM_CONFIRM = 0.3  # D6: between CONFIRM and ACCEPT → ask "did you mean …?"
_BUDGET = ("budget_min", "budget_max", "budget_target")
log = logging.getLogger(__name__)


async def understand(
    text: str,
    ctx: ExtractionContext,
    extractor: LLMSlotExtractor,
    normalizer: Normalizer,
    alias_map: Mapping[AliasKind, Mapping[str, str]],
) -> Understanding:
    usage: list[Usage] = []
    fallbacks: list[str] = []
    try:
        result = await extractor.extract(text, ctx)
        ti = result.value
        usage.append(result.usage)
        if result.usage.provider != extractor.primary_provider:
            fallbacks.append("llm_failover")
    except ProviderError as exc:
        log.warning("extractor failed, using rules: %s", exc)
        ti = rules_extract(text, alias_map)
        fallbacks.append("rules_extractor")

    updates = _slot_updates(ti, parse_numeric(text))
    low: list[LowConfidence] = []
    unsupported: list[Unsupported] = []
    mentions = (
        (AliasKind.CITY, "city", ti.slots.city_mention),
        (AliasKind.MAKE, "make", ti.slots.make_mention),
        (AliasKind.MODEL, "model", ti.slots.model_mention),
    )
    for kind, slot, mention in mentions:
        if not mention:
            continue
        m = await normalizer.normalize(kind, mention)
        if m.canonical and m.score >= TRGM_ACCEPT:
            updates[slot] = m.canonical
        elif m.canonical and m.score >= TRGM_CONFIRM:
            low.append(LowConfidence(slot=slot, raw=mention, candidate=m.canonical, score=m.score))
        else:
            unsupported.append(Unsupported(kind=kind, raw=mention))

    valid, dropped = sanitize_updates(updates)
    if dropped:
        log.warning("dropped invalid slot values: %s", dropped)
    return Understanding(turn_intent=ti, updates=valid, low_confidence=low,
                         unsupported=unsupported, usage=usage, fallbacks=fallbacks)


def _slot_updates(ti: TurnIntent, numeric: NumericSlots) -> dict[str, Any]:
    s = ti.slots
    updates: dict[str, Any] = {}
    for name in ("body_type", "fuel", "purpose"):
        value = getattr(s, name)
        if value is not None:
            updates[name] = value
    if numeric.has_budget:  # D3: the parser wins on numbers
        updates.update(budget_min=numeric.budget_min, budget_max=numeric.budget_max,
                       budget_target=numeric.budget_target)
    elif any(getattr(s, b) is not None for b in _BUDGET):
        cap = round(s.budget_target * 1.1) if s.budget_target else s.budget_max
        updates.update(budget_min=s.budget_min, budget_max=cap, budget_target=s.budget_target)
    for name in ("max_km", "min_payload_kg", "min_year"):
        value = getattr(numeric, name)
        if value is None:
            value = getattr(s, name)
        if value is not None:
            updates[name] = value
    for name in ti.clear_slots:
        if name == "budget":
            updates.update(dict.fromkeys(_BUDGET))
        else:
            updates[name] = None
    return updates
```

```python
"""Compose the spoken top-k answer: LLM phrasing from the fact sheet, validated in code (F1)."""

import json
import logging

from pydantic import BaseModel, Field

from voice_agent.domain.enums import Language
from voice_agent.domain.ports import LLM, ProviderError, Usage
from voice_agent.response import templates
from voice_agent.response.factsheet import FactSheet
from voice_agent.response.schema import ComposedResponse, Verdict, Violation
from voice_agent.response.validator import EntityVocab, validate

MAX_ATTEMPTS = 2
log = logging.getLogger(__name__)

_SYSTEM = """You write short spoken answers for a used commercial-vehicle voice assistant.
You get a FACT SHEET of vehicles in rank order. Rules:
1. Exactly one item per vehicle, in the given order, with its listing_id.
2. Use only that vehicle's own facts. Copy values exactly as written ("₹4.85 lakh", "48,000 km").
   Never compute, round or convert numbers. Never write numbers as words.
3. Mention the vehicle's reasons in plain words.
4. intro and closing: no numbers and no vehicle names. closing invites a follow-up question.
5. Each item under 30 words; sound natural when spoken.
Language: {language}."""
_LANG = {Language.EN: "English", Language.HINGLISH: "Hinglish in Roman script"}


class ComposeOutcome(BaseModel):
    response: ComposedResponse
    verdict: Verdict
    usage: list[Usage] = Field(default_factory=list)
    fallbacks: list[str] = Field(default_factory=list)
    attempts: int = 0


async def compose_results(facts: FactSheet, llm: LLM, vocab: EntityVocab) -> ComposeOutcome:
    system = _SYSTEM.format(language=_LANG[facts.language])
    usage: list[Usage] = []
    violations: list[Violation] = []
    for attempt in range(1, MAX_ATTEMPTS + 1):
        try:
            result = await llm.structured(system, _user_prompt(facts, violations), ComposedResponse)
        except ProviderError as exc:
            log.warning("composer LLM failed: %s", exc)
            break
        usage.append(result.usage)
        verdict = validate(result.value, facts, vocab)
        if verdict.ok:
            fallbacks = ["regenerated"] if attempt > 1 else []
            return ComposeOutcome(response=result.value, verdict=verdict, usage=usage,
                                  fallbacks=fallbacks, attempts=attempt)
        violations = verdict.violations
        log.warning("validator rejected attempt %d: %s", attempt, violations)
    return ComposeOutcome(
        response=templates.results(facts),
        verdict=Verdict(ok=True, violations=violations, source="template"),
        usage=usage, fallbacks=["template"], attempts=MAX_ATTEMPTS,
    )


def _user_prompt(facts: FactSheet, violations: list[Violation]) -> str:
    prompt = "FACT SHEET:\n" + json.dumps(facts.model_dump(mode="json")["items"], ensure_ascii=False)
    if violations:
        problems = "; ".join(f"{v.where}: {v.kind} {v.detail}" for v in violations)
        prompt += f"\nYour previous answer was rejected for: {problems}. Fix these."
    return prompt
```

```python
"""Stage timers for per-turn latency (J4)."""

import time
from collections.abc import Iterator
from contextlib import contextmanager


class Stopwatch:
    def __init__(self) -> None:
        self.ms = 0.0


@contextmanager
def timed() -> Iterator[Stopwatch]:
    sw = Stopwatch()
    start = time.perf_counter()
    try:
        yield sw
    finally:
        sw.ms = round((time.perf_counter() - start) * 1000, 1)
```

```python
"""LangGraph wiring (C2). Nodes call one component each and write state; no business logic here."""

import logging
from collections.abc import Mapping, Sequence
from dataclasses import dataclass
from typing import Any

from langgraph.checkpoint.base import BaseCheckpointSaver
from langgraph.graph import END, START, StateGraph

from voice_agent.catalog.filters import constraint_violations
from voice_agent.catalog.ranker import rank
from voice_agent.catalog.relax import propose_relaxation
from voice_agent.domain.enums import AliasKind, City, Language
from voice_agent.domain.models import Filters, ResultRef, Slots
from voice_agent.domain.ports import LLM, Catalog, Normalizer
from voice_agent.graph.state import ConversationState
from voice_agent.response import templates
from voice_agent.response.composer import compose_results
from voice_agent.response.factsheet import build_fact_item, build_fact_sheet
from voice_agent.response.schema import ComposedResponse, Verdict
from voice_agent.response.validator import EntityVocab
from voice_agent.state.models import Memory
from voice_agent.state.reducer import apply_understanding
from voice_agent.telemetry.timing import timed
from voice_agent.understanding.extractor import LLMSlotExtractor
from voice_agent.understanding.pipeline import understand
from voice_agent.understanding.schema import ExtractionContext

log = logging.getLogger(__name__)


@dataclass(frozen=True)
class Deps:
    llm: LLM
    normalizer: Normalizer
    catalog: Catalog
    vocab: EntityVocab
    neighbors: Mapping[City, Sequence[City]]
    weights: Mapping[str, float]
    alias_map: Mapping[AliasKind, Mapping[str, str]]


def _memory(s: ConversationState) -> Memory:
    return Memory(
        slots=s.get("slots") or Slots(),
        pending_confirmation=s.get("pending_confirmation"),
        pending_relaxation=s.get("pending_relaxation"),
        last_results=s.get("last_results") or [],
        focused_listing_id=s.get("focused_listing_id"),
    )


def _pending_question(mem: Memory) -> str | None:
    if mem.pending_confirmation:
        return f"Did you mean {mem.pending_confirmation.candidate}? (yes/no)"
    if mem.pending_relaxation:
        r = mem.pending_relaxation
        return f"Offered to relax {r.slot} to {r.new}. (relax if accepted)"
    return None


def _timing(s: ConversationState, stage: str, ms: float) -> dict[str, float]:
    return {**s.get("timings_ms", {}), stage: ms}


def build_graph(deps: Deps, checkpointer: BaseCheckpointSaver[Any] | None = None) -> Any:
    extractor = LLMSlotExtractor(deps.llm, deps.alias_map)

    async def understand_node(s: ConversationState) -> dict[str, Any]:
        mem = _memory(s)
        ctx = ExtractionContext(slots=mem.slots, last_results=mem.last_results,
                                pending_question=_pending_question(mem))
        with timed() as t:
            u = await understand(s["text"], ctx, extractor, deps.normalizer, deps.alias_map)
        return {
            "understanding": u, "timings_ms": _timing(s, "nlu", t.ms),
            "usage": [*s.get("usage", []), *u.usage],
            "fallback_used": [*s.get("fallback_used", []), *u.fallbacks],
        }

    async def update_state_node(s: ConversationState) -> dict[str, Any]:
        understanding = s["understanding"]
        assert understanding is not None
        mem, diff, route = apply_understanding(_memory(s), understanding)
        return {
            "turn": s.get("turn", 0) + 1, "slots": mem.slots,
            "pending_confirmation": mem.pending_confirmation,
            "pending_relaxation": mem.pending_relaxation,
            "last_results": mem.last_results, "focused_listing_id": mem.focused_listing_id,
            "slot_diff": diff, "route": route,
        }

    async def search_node(s: ConversationState) -> dict[str, Any]:
        filters = Filters.from_slots(s["slots"])
        with timed() as t:
            result = await deps.catalog.search(filters)
        safe = [r for r in result.rows if not constraint_violations(r, filters)]
        fallbacks = list(s.get("fallback_used", []))
        if len(safe) != len(result.rows):  # F2.3: should never happen; loud if it does
            log.error("constraint re-check dropped %d rows", len(result.rows) - len(safe))
            fallbacks.append("constraint_drop")
        return {"filters": filters, "executed_sql": result.sql, "sql_params": result.params,
                "candidates": safe, "fallback_used": fallbacks,
                "timings_ms": _timing(s, "search", t.ms)}

    async def rank_node(s: ConversationState) -> dict[str, Any]:
        with timed() as t:
            results = rank(s["candidates"], s["slots"], deps.weights)
        refs = [ResultRef(listing_id=r.listing.listing_id, rank=r.rank,
                          make=r.listing.make, model=r.listing.model) for r in results]
        return {"results": results, "last_results": refs, "focused_listing_id": None,
                "timings_ms": _timing(s, "rank", t.ms)}

    async def relax_node(s: ConversationState) -> dict[str, Any]:
        filters = s["filters"]
        assert filters is not None
        with timed() as t:
            relaxation = await propose_relaxation(filters, deps.catalog.count, deps.neighbors)
        return {"pending_relaxation": relaxation, "timings_ms": _timing(s, "relax", t.ms)}

    async def respond_node(s: ConversationState) -> dict[str, Any]:
        route = s["route"]
        assert route is not None
        understanding = s.get("understanding")
        language = understanding.turn_intent.language if understanding else Language.EN
        usage, fallbacks = list(s.get("usage", [])), list(s.get("fallback_used", []))
        verdict = Verdict(ok=True, source="template")
        response: ComposedResponse
        with timed() as t:
            if route.kind == "search" and s.get("results"):
                facts = build_fact_sheet(s["results"], s["slots"].purpose, language)
                outcome = await compose_results(facts, deps.llm, deps.vocab)
                response, verdict = outcome.response, outcome.verdict
                usage += outcome.usage
                fallbacks += outcome.fallbacks
            elif route.kind == "search":
                response = templates.zero_results(s["slots"], s.get("pending_relaxation"))
            elif route.kind == "follow_up" and route.listing_id:
                listing = await deps.catalog.get(route.listing_id)
                ranks = {r.listing_id: r.rank for r in s.get("last_results", [])}
                if listing is None or route.follow_up_field is None:
                    response = templates.simple("which_one")
                else:
                    item = build_fact_item(listing, ranks.get(listing.listing_id, 1), [], None)
                    response = templates.follow_up(item, route.follow_up_field)
            elif route.kind == "confirm" and s.get("pending_confirmation"):
                response = templates.confirm(s["pending_confirmation"])  # type: ignore[arg-type]
            elif route.kind == "unsupported":
                response = templates.unsupported(route.unsupported)
            elif route.kind in ("reset", "out_of_domain", "need_info", "which_one"):
                response = templates.simple(route.kind)
            else:
                response = templates.simple("error")
        return {"response": response, "verdict": verdict, "usage": usage,
                "fallback_used": fallbacks, "timings_ms": _timing(s, "compose", t.ms)}

    def after_state(s: ConversationState) -> str:
        route = s["route"]
        return "search" if route is not None and route.kind == "search" else "respond"

    def after_search(s: ConversationState) -> str:
        return "rank" if s["candidates"] else "relax"

    g = StateGraph(ConversationState)
    g.add_node("understand", understand_node)
    g.add_node("update_state", update_state_node)
    g.add_node("search", search_node)
    g.add_node("rank", rank_node)
    g.add_node("relax", relax_node)
    g.add_node("respond", respond_node)
    g.add_edge(START, "understand")
    g.add_edge("understand", "update_state")
    g.add_conditional_edges("update_state", after_state, ["search", "respond"])
    g.add_conditional_edges("search", after_search, ["rank", "relax"])
    g.add_edge("rank", "respond")
    g.add_edge("relax", "respond")
    g.add_edge("respond", END)
    return g.compile(checkpointer=checkpointer)
```

```python
"""TurnRunner: the seam the CLI, eval and (Plan 2) API call. One call = one conversational turn."""

from pathlib import Path
from typing import Any

from pydantic import BaseModel, Field

from voice_agent.domain.models import Relaxation, ScoredListing
from voice_agent.domain.ports import Usage
from voice_agent.graph.state import TURN_RESET
from voice_agent.response.schema import ComposedResponse, Verdict
from voice_agent.state.models import PendingConfirmation
from voice_agent.telemetry.timing import timed
from voice_agent.telemetry.turnlog import append_turn


class TurnResult(BaseModel):
    session_id: str
    turn: int
    text: str
    intent: str | None
    language: str | None
    route: str
    slots: dict[str, Any]
    slot_diff: dict[str, list[Any]]
    filters: dict[str, Any] | None
    sql: str | None
    sql_params: dict[str, Any]
    results: list[ScoredListing]
    relaxation: Relaxation | None
    pending_confirmation: PendingConfirmation | None
    followup_listing_id: str | None
    followup_rank: int | None
    response: ComposedResponse
    response_text: str
    verdict: Verdict
    fallback_used: list[str]
    timings_ms: dict[str, float]
    usage: list[Usage] = Field(default_factory=list)


class TurnRunner:
    def __init__(self, graph: Any, turn_log_path: Path | None) -> None:
        self._graph = graph
        self._log = turn_log_path

    async def run(self, session_id: str, text: str) -> TurnResult:
        with timed() as total:
            state = await self._graph.ainvoke(
                {**TURN_RESET, "text": text},
                config={"configurable": {"thread_id": session_id}},
            )
        result = _to_result(session_id, text, state, total.ms)
        if self._log is not None:
            append_turn(self._log, result.model_dump(mode="json"))
        return result


def _to_result(session_id: str, text: str, s: dict[str, Any], total_ms: float) -> TurnResult:
    route = s["route"]
    u = s.get("understanding")
    is_follow_up = route.kind == "follow_up"
    ranks = {r.listing_id: r.rank for r in s.get("last_results", [])}
    response = s["response"]
    return TurnResult(
        session_id=session_id, turn=s.get("turn", 0), text=text,
        intent=u.turn_intent.intent.value if u else None,
        language=u.turn_intent.language.value if u else None,
        route=route.kind, slots=s["slots"].filled(),
        slot_diff={k: list(v) for k, v in s.get("slot_diff", {}).items()},
        filters=s["filters"].as_dict() if s.get("filters") else None,
        sql=s.get("executed_sql"), sql_params=s.get("sql_params", {}),
        results=s.get("results", []),
        relaxation=s.get("pending_relaxation") if route.kind == "search" and not s.get("results") else None,
        pending_confirmation=s.get("pending_confirmation"),
        followup_listing_id=route.listing_id if is_follow_up else None,
        followup_rank=ranks.get(route.listing_id) if is_follow_up and route.listing_id else None,
        response=response, response_text=response.text(), verdict=s["verdict"],
        fallback_used=s.get("fallback_used", []),
        timings_ms={**s.get("timings_ms", {}), "total": total_ms},
        usage=s.get("usage", []),
    )
```

```python
from collections.abc import AsyncIterator

@asynccontextmanager
async def build_runtime(settings: Settings, log_turns: bool = True) -> AsyncIterator[TurnRunner]:
    async with AsyncConnectionPool(settings.database_url, open=False,
                                   kwargs={"autocommit": True}) as pool:
        await seed(pool, settings.data_dir)
        deps = Deps(
            llm=make_llm(settings), normalizer=PgNormalizer(pool), catalog=PgCatalog(pool),
            vocab=await load_vocab(pool), neighbors=await load_neighbors(pool),
            weights=load_weights(settings.ranking_config), alias_map=await load_alias_map(pool),
        )
        async with AsyncPostgresSaver.from_conn_string(settings.database_url) as saver:
            await saver.setup()
            log_path = settings.turn_log_path if log_turns else None
            yield TurnRunner(build_graph(deps, saver), log_path)
```

```python
"""Eval harness (component 7): run eval/cases.yaml through the real pipeline and score it.

Usage: uv run python eval/run_eval.py [--cases eval/cases.yaml]
Needs Postgres (docker compose up -d db) and GROQ_API_KEY. TTS/STT are not used here.
"""

import argparse
import json
import statistics
from collections import defaultdict
from datetime import UTC, datetime
from pathlib import Path
from typing import Any
from uuid import uuid4

import yaml

from voice_agent.aio import run
from voice_agent.catalog.filters import constraint_violations
from voice_agent.config import get_settings
from voice_agent.domain.models import Filters
from voice_agent.graph.runner import TurnResult
from voice_agent.graph.runtime import build_runtime

RESULTS_DIR = Path(__file__).parent / "results"


def check_turn(expect: dict[str, Any], r: TurnResult) -> list[str]:
    failures: list[str] = []
    filters = r.filters or {}
    if "expect_filters" in expect and filters != expect["expect_filters"]:
        failures.append(f"filters {filters} != {expect['expect_filters']}")
    if "expect_changed" in expect and sorted(r.slot_diff) != sorted(expect["expect_changed"]):
        failures.append(f"changed {sorted(r.slot_diff)} != {sorted(expect['expect_changed'])}")
    if "expect_changed_count" in expect and len(r.slot_diff) != expect["expect_changed_count"]:
        failures.append(f"changed {len(r.slot_diff)} slots, expected {expect['expect_changed_count']}")
    if "expect_route" in expect and r.route != expect["expect_route"]:
        failures.append(f"route {r.route} != {expect['expect_route']}")
    if "expect_results" in expect and len(r.results) != expect["expect_results"]:
        failures.append(f"{len(r.results)} results != {expect['expect_results']}")
    if "expect_results_min" in expect and len(r.results) < expect["expect_results_min"]:
        failures.append(f"{len(r.results)} results < {expect['expect_results_min']}")
    if "expect_relaxation" in expect and (r.relaxation is not None) != expect["expect_relaxation"]:
        failures.append(f"relaxation offered={r.relaxation is not None}")
    if "expect_followup_rank" in expect and r.followup_rank != expect["expect_followup_rank"]:
        failures.append(f"follow-up rank {r.followup_rank} != {expect['expect_followup_rank']}")
    if r.route == "search" and r.results and (r.verdict.source != "llm" or "regenerated" in r.fallback_used):
        failures.append(f"validator not clean on first try ({r.verdict.source}, {r.fallback_used})")
    return failures


def safety_violations(r: TurnResult) -> int:
    if not r.filters:
        return 0
    f = Filters.model_validate(r.filters)
    return sum(1 for s in r.results if constraint_violations(s.listing, f))


async def run_cases(cases: list[dict[str, Any]]) -> dict[str, Any]:
    report: list[dict[str, Any]] = []
    timings: dict[str, list[float]] = defaultdict(list)
    shown = unsafe = 0
    async with build_runtime(get_settings(), log_turns=False) as runner:
        for case in cases:
            session = f"eval-{case['id']}-{uuid4().hex[:6]}"
            failures: list[str] = []
            for i, turn in enumerate(case["turns"], start=1):
                r = await runner.run(session, turn["say"])
                failures += [f"t{i}: {f}" for f in check_turn(turn, r)]
                shown += len(r.results)
                unsafe += safety_violations(r)
                for stage, ms in r.timings_ms.items():
                    timings[stage].append(ms)
            report.append({"id": case["id"], "category": case["category"],
                           "passed": not failures, "failures": failures})
            print(f"{'PASS' if not failures else 'FAIL'}  {case['id']:<16} {'; '.join(failures)}")
    return {"cases": report, "timings": timings, "shown": shown, "unsafe": unsafe}


def summarise(result: dict[str, Any]) -> str:
    cases = result["cases"]
    by_cat: dict[str, list[bool]] = defaultdict(list)
    for c in cases:
        by_cat[c["category"]].append(c["passed"])
    passed = sum(c["passed"] for c in cases)
    lines = [f"# Eval results ({datetime.now(UTC):%Y-%m-%d %H:%M} UTC)", "",
             f"**Overall: {passed}/{len(cases)} = {100 * passed / len(cases):.0f}%**", "",
             "| Category | Passed |", "|---|---|"]
    lines += [f"| {cat} | {sum(v)}/{len(v)} |" for cat, v in sorted(by_cat.items())]
    safe_pct = 100 * (1 - result["unsafe"] / result["shown"]) if result["shown"] else 100.0
    lines += ["", f"Constraint safety: {safe_pct:.1f}% of {result['shown']} shown results "
                  "satisfy every hard filter.", "", "| Stage | p50 ms | p95 ms |", "|---|---|---|"]
    for stage, values in sorted(result["timings"].items()):
        ordered = sorted(values)
        p95 = ordered[min(len(ordered) - 1, int(0.95 * len(ordered)))]
        lines.append(f"| {stage} | {statistics.median(ordered):.0f} | {p95:.0f} |")
    failed = [c for c in cases if not c["passed"]]
    if failed:
        lines += ["", "## Failures", ""] + [f"- `{c['id']}`: {'; '.join(c['failures'])}" for c in failed]
    return "\n".join(lines) + "\n"


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--cases", type=Path, default=Path(__file__).parent / "cases.yaml")
    args = parser.parse_args()
    cases = yaml.safe_load(args.cases.read_text(encoding="utf-8"))
    result = run(run_cases(cases))
    summary = summarise(result)
    print("\n" + summary)
    RESULTS_DIR.mkdir(exist_ok=True)
    (RESULTS_DIR / "latest.md").write_text(summary, encoding="utf-8")
    (RESULTS_DIR / "latest.json").write_text(json.dumps(result, indent=2), encoding="utf-8")


if __name__ == "__main__":
    main()
```
